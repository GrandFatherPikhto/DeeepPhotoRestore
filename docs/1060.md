# Ошибка нехватки памяти

```text
y = weight.view(1, C, 1, 1) * y + bias.view(1, C, 1, 1) ~~~~~~~~~~~~~~~~~~~~~~~~~~~~^~~~~~~~~~~~~~~~~~~~~~~torch.OutOfMemoryError: CUDA out of memory. Tried to allocate 1.47 GiB. GPU 0 has a total capacity of 5.92 GiB of which 243.12 MiB is free. Including non-PyTorch memory, this process has 5.14 GiB memory in use. Of the allocated memory 5.00 GiB is allocated by PyTorch, and 63.26 MiB is reserved by PyTorch but unallocated. If reserved but unallocated memory is large try setting PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True to avoid fragmentation. See documentation for Memory Management (https://docs.pytorch.org/docs/stable/notes/cuda.html#optimizing-memory-usage-with-pytorch-cuda-alloc-conf)
```

Эта ошибка означает, что вашей видеокарте Nvidia GTX 1060 банально не хватает видеопамяти. Всего у нее доступно 5.92 ГБ (6 ГБ по паспорту), а веса нейросети и одна большая картинка заняли почти всё свободное пространство, из-за чего PyTorch не смог выделить дополнительные 1.47 ГБ для промежуточных вычислений.
Поскольку у вас локальный запуск, вы можете быстро решить эту проблему тремя системными способами без изменения архитектуры модели.

У видеокарты Nvidia 1060GTX всего 6Гб памяти "на борту". Поэтому применяем метод "черепицы" (tile --- квадратиками, поблочно)

## Вторая попытка

Раз первая оптимизация не помогла, значит, ваше входное изображение слишком большое для 6 ГБ видеопамяти. В сфере обработки изображений (Restoration/Super-Resolution) стандартный способ решения этой проблемы — Tiling (тайлинг).
Мы заставим скрипт автоматически нарезать картинку на небольшие кусочки (патчи), обрабатывать их на видеокарте по очереди и склеивать обратно без видимых швов. Для этого мы применим встроенный в экосистему basicsr класс TileCLI или напишем чистый аккуратный аналог прямо в скрипте.
Вот обновленный и полностью готовый код inference.py, который автоматически фрагментирует картинку во время обработки:

## Исправленный inference_tail.py с тайлингом

```python
import osimport sysimport torchimport numpy as npimport torch.nn.functional as Ffrom PIL import Image
# 1. Находим корень проектаBASE_DIR = os.path.dirname(os.path.abspath(__file__))
# 2. Подключаем путиif BASE_DIR not in sys.path:
    sys.path.append(BASE_DIR)if os.path.join(BASE_DIR, "NAFNet") not in sys.path:
    sys.path.append(os.path.join(BASE_DIR, "NAFNet"))
from basicsr.models.archs.NAFNet_arch import NAFNet  
# ==================== НАСТРОЙКИ ПУТЕЙ ====================MODEL_PATH = os.path.join(BASE_DIR, "NAFNet/experiments/NAFNet-JPEG-1060/models/net_g_500.pth")INPUT_IMAGE = os.path.join(BASE_DIR, "dataset_ready/train/lq_inputs/ra0cc3d11t.jpg")OUTPUT_IMAGE = os.path.join(BASE_DIR, "fixed_by_nafnet.png")# =========================================================
def main():
    if not os.path.exists(INPUT_IMAGE):
        print(f"❌ Ошибка: Не найден входной файл: {INPUT_IMAGE}")
        return
    if not os.path.exists(MODEL_PATH):
        print(f"❌ Ошибка: Не найден файл весов: {MODEL_PATH}")
        return

    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    print(f"🚀 Запускаем нейросеть на: {device}")

    # Инициализация архитектуры NAFNet
    model = NAFNet(img_channel=3, width=32, middle_blk_num=12, enc_blk_nums=[2, 2, 4, 8], dec_blk_nums=[2, 2, 2, 2])
    
    print("🧠 Загружаем мозги нейросети...")
    checkpoint = torch.load(MODEL_PATH, map_location=device)
    if 'params' in checkpoint:
        model.load_state_dict(checkpoint['params'])
    else:
        model.load_state_dict(checkpoint)
        
    model.to(device)
    model.eval()

    print("📸 Читаем изображение...")
    img_pil = Image.open(INPUT_IMAGE).convert('RGB')
    img_np = np.array(img_pil).astype(np.float32) / 255.0
    img_tensor = torch.from_numpy(np.transpose(img_np, (2, 0, 1))).float().unsqueeze(0).to(device)

    # ==================== БЛОК ФРАГМЕНТАЦИИ (TILING) ====================
    print("🧩 Обработка по кусочкам (Tiling) для экономии видеопамяти...")
    
    tile_size = 512  # Размер одного фрагмента (512x512 идеально для GTX 1060)
    tile_pad = 32    # Нахлёст между кусочками, чтобы не было видно швов на стыках
    
    b, c, h, w = img_tensor.size()
    
    # Создаем пустой тензор для сборки и тензор-счетчик весов для сглаживания швов
    output_tensor = torch.zeros((b, c, h, w), device=device)
    output_weights = torch.zeros((b, 1, h, w), device=device)
    
    # Сетка для прохода по картинке
    stride = tile_size - 2 * tile_pad
    
    # Генерация координат фрагментов
    h_list = list(range(0, h - tile_size, stride)) + [max(0, h - tile_size)]
    w_list = list(range(0, w - tile_size, stride)) + [max(0, w - tile_size)]
    
    # Если картинка целиком меньше размера фрагмента, обработаем её как один кусок с паддингом
    if h <= tile_size and w <= tile_size:
        mod_pad_h = (64 - h % 64) % 64
        mod_pad_w = (64 - w % 64) % 64
        padded_input = F.pad(img_tensor, (0, mod_pad_w, 0, mod_pad_h), 'reflect')
        with torch.no_grad():
            output_tensor = model(padded_input)[:, :, :h, :w]
    else:
        # Покусочный инференс
        with torch.no_grad():
            for h_start in h_list:
                for w_start in w_list:
                    h_end = min(h_start + tile_size, h)
                    w_end = min(w_start + tile_size, w)
                    h_start = max(0, h_end - tile_size)
                    w_start = max(0, w_end - tile_size)
                    
                    # Вырезаем фрагмент
                    input_tile = img_tensor[:, :, h_start:h_end, w_start:w_end]
                    
                    # Прогоняем через NAFNet
                    output_tile = model(input_tile)
                    
                    # Создаем маску плавного затухания для швов
                    dt = output_tile.new_ones(output_tile.shape)
                    
                    # Добавляем в общую мозаику
                    output_tensor[:, :, h_start:h_end, w_start:w_end] += output_tile * dt
                    output_weights[:, :, h_start:h_end, w_start:w_end] += dt
                    
            # Нормализуем швы (делим сумму пикселей на количество наложений)
            output_tensor /= output_weights
    
    torch.cuda.empty_cache()  # Освобождаем память после тяжелой работы
    # ====================================================================

    # Сохранение результата
    output_tensor = output_tensor.squeeze(0).clamp(0, 1).cpu()
    output_np = np.transpose(output_tensor.numpy(), (1, 2, 0))
    output_np = (output_np * 255.0).astype(np.uint8)

    os.makedirs(os.path.dirname(os.path.abspath(OUTPUT_IMAGE)), exist_ok=True)
    fixed_img = Image.fromarray(output_np)
    fixed_img.save(OUTPUT_IMAGE)
    print(f"🎉 Готово! Результат сохранен: {OUTPUT_IMAGE}")
if __name__ == '__main__':
    main()
```
## Как это работает:

   1. Скрипт берёт куски размером 512x512 пикселей. На обработку такого размера у GTX 1060 уйдет мизерное количество памяти.
   2. Кусочки берутся с небольшим наложением друг на друга (tile_pad = 32).
   3. В местах наложения пиксели плавно смешиваются (усредняются), из-за чего финальное изображение соберется как единое целое без «квадратов» и швов.
