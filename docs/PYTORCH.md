# """Скрипт верификации системных параметров PyTorch и бенчмаркинга GPU."""

import sysimport timeimport torch
def run_benchmark():
    print("=" * 50)
    print(" СИСТЕМНАЯ ИНФОРМАЦИЯ ".center(50, "="))
    print("=" * 50)
    print(f"Версия Python:       {sys.version.split()[0]}")
    print(f"Версия PyTorch:      {torch.__version__}")

    cuda_available = torch.cuda.is_available()
    print(f"Доступность CUDA:    {'ВКЛЮЧЕНА' if cuda_available else 'ВЫКЛЮЧЕНА'}")

    if not cuda_available:
        print("\n[ОШИБКА]: Доступ к GPU отсутствует. Вычисления невозможны.")
        return

    device_properties = torch.cuda.get_device_properties(0)
    total_memory_gb = device_properties.total_memory / (1024 ** 3)
    
    print(f"Версия CUDA в Torch: {torch.version.cuda}")
    print(f"Имя устройства:      {torch.cuda.get_device_name(0)}")
    print(f"Память GPU (всего):  {total_memory_gb:.2f} GB")

    print("\n" + "=" * 50)
    print(" ТЕСТ ПРОИЗВОДИТЕЛЬНОСТИ ".center(50, "="))
    print("=" * 50)

    # Инициализация тензоров на графическом процессоре
    device = torch.device("cuda")
    x = torch.randn(10000, 10000, device=device, dtype=torch.float32)
    y = torch.randn(10000, 10000, device=device, dtype=torch.float32)

    # Предварительный запуск (Warm-up) для исключения времени на аллокацию памяти
    _ = torch.matmul(x, y)
    torch.cuda.synchronize()

    # Фиксация времени выполнения операции матричного умножения
    start_time = time.time()
    _ = torch.matmul(x, y)
    torch.cuda.synchronize()
    end_time = time.time()

    execution_time_ms = (end_time - start_time) * 1000
    print(f"Время выполнения операции (Мат. умножение): {execution_time_ms:.2f} мс")
    print("=" * 50)
if __name__ == "__main__":
    run_benchmark()
```

## 6. Модуль генерации входных данных (Имитация артефактов сжатия JPEG)


Для обучения и тестирования нейросетевых моделей реставрации графики требуется контролируемая деградация исходных изображений высокого качества (Ground Truth). Данный скрипт автоматизирует процесс массового сжатия исходных файлов в формат JPEG с критически низким коэффициентом качества, имитируя сильные артефакты блочности и потерю высокочастотных деталей.

```python
#!/usr/bin/env python3
#-*- coding: utf-8 -*-
"""
Модуль пакетной деградации изображений методом агрессивного JPEG-сжатия.
"""
import osfrom pathlib import Pathimport cv2
# Конфигурация путей файловой системыSOURCE_DIR = Path.home() / "Images" / "jpeg"DISTORTED_DIR = Path.home() / "Images" / "compressed"
# Инициализация целевой директории при ее отсутствии
DISTORTED_DIR.mkdir(parents=True, exist_ok=True)
# Параметр качества кодирования JPEG (Диапазон: 0–100, где 0 — максимальная деградация)JPEG_QUALITY = 10
```python
def distort_images():
    """
    Выполняет пакетное чтение изображений, применение артефактов сжатия
    и запись результатов в целевой каталог.
    """
    # Сбор путей к файлам с допустимыми расширениями (без учета регистра)
    extensions = {".jpg", ".jpeg", ".PNG", ".png"}  # Оптимизировано для поддержки разных форматов
    image_files = [f for f in SOURCE_DIR.iterdir() if f.suffix.lower() in extensions]

    if not image_files:
        print(f"[ВНИМАНИЕ]: В директории {SOURCE_DIR} отсутствуют поддерживаемые файлы.")
        return

    print(f"[СТАРТ]: Обнаружено файлов: {len(image_files)}. Запуск процесса деградации...")

    processed_count = 0
    for img_path in image_files:
        # Чтение матрицы пикселей из исходного файла
        img = cv2.imread(str(img_path))
        if img is None:
            print(f"[ОШИБКА]: Не удалось прочитать файл: {img_path.name}")
            continue

        # Формирование пути сохранения целевого файла
        output_path = DISTORTED_DIR / img_path.name

        # Экспорт изображения с жестко заданным параметром компрессии
        success = cv2.imwrite(
            str(output_path), 
            img, 
            [int(cv2.IMWRITE_JPEG_QUALITY), JPEG_QUALITY]
        )
        
        if success:
            processed_count += 1
            print(f"[ОБРАБОТКА]: {img_path.name} -> Экспортировано в {DISTORTED_DIR.name}")

    print(f"\n[УСПЕХ]: Обработка завершена. Успешно сжато файлов: {processed_count}/{len(image_files)}.")

if __name__ == "__main__":
    distort_images()
```

Из текста удалены разговорные маркеры, логирование структурировано в соответствии со стандартами корпоративного ПО, а промежуточные шаги конвейера (pipeline) описаны с применением точной терминологии компьютерного зрения и глубокого обучения.

## 7. Модуль инференса и суперразрешения (Восстановление изображений)
Данный скрипт реализует конвейер постобработки деградированных изображений. Программа считывает сжатые файлы, выполняет нормализацию, изменяет порядок осей тензоров для соответствия стандартам PyTorch [B, C, H, W], переносит вычисления на графический ускоритель и применяет математическую интерполяцию для деблокинга и масштабирования. Вывод потока данных принудительно дебуферизирован (flush=True) для обеспечения корректного мониторинга в WSL2.

```python
#!/usr/bin/env python3
#-*- coding: utf-8 -*-
"""
Модуль инференса: масштабирование и подавление артефактов JPEG-сжатия.
"""
import osimport sysfrom pathlib import Pathimport cv2import torch
# Принудительное отключение буферизации для синхронизации вывода в консоль WSL2
print("[ИНФО]: Инициализация скрипта инференса. Валидация директорий...", flush=True)
INPUT_DIR = Path.home() / "Images" / "compressed"OUTPUT_DIR = Path.home() / "Images" / "restored"
OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
if not INPUT_DIR.exists():
    print(
        f"[КРИТИЧЕСКАЯ ОШИБКА]: Исходная директория отсутствует: {INPUT_DIR}",
        file=sys.stderr,
        flush=True,
    )
    sys.exit(1)
# Сбор целевых файлов (поддерживаемые расширения)image_files = list(INPUT_DIR.glob("*.jpg")) + list(INPUT_DIR.glob("*.jpeg"))
print(f"[ИНФО]: Обнаружено файлов для обработки: {len(image_files)}", flush=True)
if not image_files:
    print("[ВНИМАНИЕ]: Целевые изображения не обнаружены. Завершение работы.", flush=True)
    sys.exit(0)

print("[ИНФО]: Проверка аппаратной конфигурации...", flush=True)device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"[ИНФО]: Активное вычислительное устройство: {device}", flush=True)
# Превентивная инициализация контекста CUDA для исключения задержек при первом вызовеif device.type == "cuda":
    print("[ИНФО]: Инициализация подсистемы CUDA...", flush=True)
    torch.cuda.init()
    print("[ИНФО]: Подсистема CUDA успешно активирована.", flush=True)

print("[ИНФО]: Запуск конвейера пакетной обработки...", flush=True)
for i, img_path in enumerate(image_files, 1):
    print(
        f"\n[{i}/{len(image_files)}] Чтение графического файла: {img_path.name}...",
        flush=True,
    )

    img = cv2.imread(str(img_path))
    if img is None:
        print(f"  └─ [ОШИБКА]: Сбой декодирования файла: {img_path.name}", flush=True)
        continue

    # Конвертация цветового пространства (BGR -> RGB) и нормализация к диапазону [0.0, 1.0]
    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    tensor_cpu = torch.from_numpy(img_rgb).float() / 255.0

    # Перенос тензора в видеопамять графического ускорителя
    tensor_gpu = tensor_cpu.to(device)

    # Транспонирование осей: [H, W, C] -> [C, H, W] и добавление батч-измерения -> [1, C, H, W]
    tensor_gpu = tensor_gpu.permute(2, 0, 1).unsqueeze(0)

    # Выполнение математических трансформаций в режиме без расчета градиентов
    with torch.no_grad():
        # Подавление высокочастотных артефактов методом бикубической интерполяции (Scale Factor = 2)
        restored_gpu = torch.nn.functional.interpolate(
            tensor_gpu, scale_factor=2, mode="bicubic", align_corners=False
        )

    # Обратное преобразование: удаление батч-измерения, транспонирование осей и денормализация
    res_cpu = restored_gpu.squeeze(0).permute(1, 2, 0).clamp(0, 1).cpu()
    res_numpy = (res_cpu.numpy() * 255).astype("uint8")
    
    # Возврат в цветовое пространство BGR для записи средствами OpenCV
    res_bgr = cv2.cvtColor(res_numpy, cv2.COLOR_RGB2BGR)

    # Экспорт восстановленного изображения
    out_path = OUTPUT_DIR / f"restored_{img_path.name}"
    cv2.imwrite(str(out_path), res_bgr)
    print(f"  └─ [УСПЕХ]: Файл сохранен -> {out_path.name}", flush=True)

print(f"\n[ЗАВЕРШЕНО]: Процесс обработки завершен. Результаты экспортированы в: {OUTPUT_DIR}", flush=True)
```

## Раздел: Архитектура и анализ фреймворка Real-ESRGAN## 1. Общие сведения и назначение пакета
realesrgan представляет собой официальную open-source библиотеку, разработанную Исследовательским центром Tencent (Applied Research Center, Tencent PCG) [1]. Пакет является высокоуровневой абстракцией над архитектурой генеративно-состязательной нейросети Real-ESRGAN (Enhanced Super-Resolution Generative Adversarial Networks) [1].
Использование данного пакета исключает необходимость ручной низкоуровневой реализации слоев нейросети, предоставляя унифицированный интерфейс для инференса моделей сверхвысокого разрешения (Super-Resolution) и реставрации графических данных [1].

## 2. Функциональные возможности и решаемые задачи## Подавление артефактов компрессии (JPEG Deblocking)
Модель эффективно идентифицирует и устраняет пространственные артефакты блочности, размытия и цветового шума, возникающие при дискретном косинусном преобразовании в алгоритмах сжатия JPEG [1].
## Генеративное сверхразрешение (Generative Super-Resolution)
Пакет выполняет масштабирование изображений (коэффициенты х2, х4 и более) с генерацией высокочастотных деталей. В отличие от классической интерполяции, нейросеть осуществляет субпиксельный синтез текстур (микрорельеф кожи, волосяной покров, текстура тканей, геометрия граней), восстанавливая утерянную информацию на основе выученного априорного распределения.
## Реставрация портретных областей (Face Restoration)
Конвейер поддерживает нативную интеграцию со специализированными слепыми моделями восстановления лиц (Blind Face Restoration), такими как GFPGAN и CodeFormer. Алгоритм локализует лицевые ориентиры (landmarks) и реконструирует геометрию глаз, губ и мимических зон из низкокачественных пиксельных паттернов.
------------------------------
## 3. Компонентный состав и архитектурные зависимости
При развертывании пакета (pip install realesrgan) в систему интегрируются следующие базовые модули:

* Класс RealESRGANer: Главный управляющий конвейер (Pipeline Manager). Он инкапсулирует логику инференса: автоматическое разбиение изображений сверхвысокого разрешения на перекрывающиеся фрагменты (тайлинг) при дефиците видеопамяти (VRAM), перенос тензоров между Host (CPU) и Device (GPU), и последующую бесшовную сборку обработанных фрагментов.
* Репозиторий предобученных весов (Pre-trained Models):
* RealESRGAN_x4plus: Базовая полноразмерная модель для обработки комплексных текстур на реальных фотографиях.
   * RealESRGAN_x4plus_anime_6B: Облегченная архитектура из 6 остаточных блоков, оптимизированная для работы с рисованным контентом (аниме, иллюстрации, схемы).
* Зависимость basicsr: Низкоуровневый фреймворк Open-Source BasicSR (Basic Super Resolution), реализующий базовые сверточные слои, блоки внимания и математические примитивы восстановления изображений.

## 4. Концептуальное отличие от классических моделей
Предшествующие модели (например, оригинальный ESRGAN) обучались на синтетических парах данных, где деградация формировалась простым изотропным уменьшением разрешения (Downsampling). Такие модели демонстрировали низкую обобщающую способность на реальных данных.
Авторы Real-ESRGAN внедрили схему деградации высокого порядка (High-Order Degradation Process) [1]. Модифицированный процесс формирования обучающей выборки включает последовательное многократное наложение случайных ядер размытия, различных типов шумов (гауссов, пуассоновский) и циклов сжатия JPEG [1]. Благодаря этому подходу ИИ устойчив к комплексным хаотичным искажениям, характерным для изображений из сети Интернет [1].

Вот очищенная и переведенная в строгий технический стиль версия раздела по устранению ошибок развертывания и ручной конфигурации весов модели. Из текста удалена разговорная лексика, упорядочена техническая терминология и зафиксированы точные причинно-следственные связи сбоев в legacy-коде.

## Раздел: Устранение ошибок совместимости и конфигурация весов (Инжиниринг среды)## 1. Разрешение конфликта совместимости BasicSR с актуальными версиями PyTorch
При развертывании пакета realesrgan в современных средах выполнения (включая сборки PyTorch под CUDA 13.x) возникает критическая ошибка инициализации на этапе импорта модулей библиотеки basicsr.
## Причина сбоя (Root Cause)
Внутренний модуль деградации данных basicsr/data/degradations.py содержит статическую директиву импорта устаревшего пространства имен torchvision.transforms.functional_tensor. В актуальных релизах фреймворка PyTorch данный низкоуровневый компонент полностью удален, а его функции перенесены в унифицированный модуль torchvision.transforms.functional.
## Инженерное решение (Патчинг исходного кода)
Для восстановления работоспособности зависимостей без понижения версии PyTorch применяется метод потокового редактирования исходного кода установленного пакета. Выполните команду sed в терминале при активированном виртуальном окружении (.venv):

```bash
sed -i 's/functional_tensor/functional/g' .venv/lib/python3.12/site-packages/basicsr/data/degradations.py
```

Механизм действия: Команда выполняет глобальную замену (/g) строкового литерала functional_tensor на functional непосредственно в файле деградаций внутри директории site-packages текущего окружения.
------------------------------
## 2. Устранение ошибки инициализации класса RealESRGANer (Валидация весов)## Причина сбоя (Root Cause)
Вызов конструктора класса RealESRGANer со значением аргумента model_path=None (стандартный паттерн для автоматической загрузки контрольных точек во многих фреймворках) в данной реализации вызывает исключение AttributeError. Внутренний парсер класса выполняет валидацию расширения файла через метод строковых операций .startswith(), который не обрабатывает тип NoneType. Пакет требует явного указания локального пути к бинарному файлу весов.
## Пошаговый алгоритм исправления

## Шаг 2. Модификация конфигурации конструктора пайплайна
В скрипте инференса (например, restore_with_ai.py) измените параметры инициализации экземпляра RealESRGANer, передав строковый путь к загруженному файлу .pth и активировав режим вычислений с пониженной точностью для оптимизации VRAM.
Исходная некорректная конфигурация:

```python
upscaler = RealESRGANer(
    scale=4,
    model_path=None,  # Вызывает исключение AttributeError
    model=model_architecture,
    ...
)
```

Целевая корректная конфигурация:

```python
upscaler = RealESRGANer(
    scale=4,
    model_path="RealESRGAN_x4plus.pth",  # Явное указание локального пути к весам
    model=model_architecture,
    tile=0,        # Размер тайла (0 — обработка изображения целиком)
    tile_pad=10,   # Поля перекрытия тайлов для предотвращения швов
    pre_pad=0,     # Предварительный паддинг на границах
    half=True,     # Активация режима FP16 (смешанная точность для тензорных ядер RTX 4070 Ti)
    device=device  # Вычислительное устройство (cuda)
)
```
## Результат выполнения конвейера

   1. Инициализация: Конструктор успешно считывает строку пути, инициализирует пустую граф-архитектуру RRDBNet в оперативной памяти (RAM) и прошивает в нее коэффициенты тензоров из файла .pth.
   2. Аллокация ресурсов: Вызов внутреннего метода переноса отправляет скомпилированную математическую модель в видеопамять графического ускорителя (VRAM).
   3. Вычисления: Активация флага half=True задействует аппаратные тензорные ядра архитектуры Ada Lovelace для ускоренного инференса, выполняя субпиксельный синтез текстур и деблокинг JPEG без задержек ввода-вывода.

Вот очищенная и переведенная в строгий академический и технический стиль финальная версия интеграционного скрипта инференса нейросети. Код структурирован, снабжен профессиональным логированием и комментариями, описывающими низкоуровневые процессы обработки тензоров на GPU.

## 8. Модуль инференса нейросети Real-ESRGAN (restore_with_ai.py)
Данный скрипт реализует полноценный конвейер генеративного восстановления и сверхразрешения (Super-Resolution). Вместо математической интерполяции здесь задействована глубокая сверточная нейросеть с архитектурой RRDBNet (Residual in Residual Dense Block).
Для предотвращения переполнения видеопамяти графического ускорителя (VRAM) применен механизм тайлинга (tile=400), разбивающий изображение на перекрывающиеся фрагменты. Параметр вычислений со смешанной точностью (half=True) оптимизирует инференс за счет использования тензорных ядер графического процессора RTX 4070 Ti.

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Финальный модуль ИИ-реставрации изображений с использованием архитектуры Real-ESRGAN.\
"""
import osimport sysfrom pathlib import Pathimport cv2import torchfrom realesrgan import RealESRGANerfrom basicsr.archs.rrdbnet_arch import RRDBNet
# Конфигурация путей ввода-вывода файловой системыINPUT_DIR = Path.home() / "Images" / "compressed"OUTPUT_DIR = Path.home() / "Images" / "restored_ai"
OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

print("[ИНФО]: Инициализация конвейера глубокого обучения...", flush=True)
# Верификация и выбор целевого вычислительного устройстваdevice = torch.device("cuda" if torch.cuda.is_available() else "cpu")if device.type != "cuda":
    print(
        "[ВНИМАНИЕ]: Графический ускоритель с поддержкой CUDA не обнаружен. "
        "Вычисления на CPU приведут к критическому падению производительности!", 
        flush=True
    )
# Декларация структуры нейросети (Классическая конфигурация RealESRGAN_x4plus)# num_in_ch/num_out_ch - каналы RGB, num_block - количество остаточных блоков плотностиmodel_architecture = RRDBNet(
    num_in_ch=3, 
    num_out_ch=3, 
    num_feat=64, 
    num_block=23, 
    num_grow_ch=32, 
    scale=4
)
try:
    # Инициализация управляющего менеджера инференса
    upscaler = RealESRGANer(
        scale=4,                            # Базовый масштаб весов модели
        model_path="RealESRGAN_x4plus.pth", # Локальный путь к скомпилированным весам
        model=model_architecture,           # Архитектурный граф сети
        tile=400,                           # Размер обрабатываемого фрагмента (патча) в пикселях
        tile_pad=10,                        # Поля перекрытия фрагментов для подавления швов сборки
        pre_pad=0,                          # Предварительное расширение границ кадра
        half=True,                          # Активация режима вычислений FP16 (смешанная точность)
        device=device                       # Вычислительный контекст устройства
    )except Exception as e:
    print(f"[КРИТИЧЕСКАЯ ОШИБКА]: Сбой инициализации графа нейросети: {e}", file=sys.stderr)
    sys.exit(1)
# Сбор и лексикографическая сортировка исходных файловimage_files = sorted(list(INPUT_DIR.glob("*.jpg")) + list(INPUT_DIR.glob("*.jpeg")))
print(f"[ИНФО]: Обнаружено деградированных файлов для реставрации: {len(image_files)}", flush=True)
if not image_files:
    print("[ИНФО]: Целевые объекты обработки отсутствуют. Конвейер остановлен.", flush=True)
    sys.exit(0)

print("[ИНФО]: Запуск процесса генеративного субпиксельного синтеза...", flush=True)
for i, img_path in enumerate(image_files, 1):
    print(f"[{i}/{len(image_files)}] Инференс файла: {img_path.name} -> ", end="", flush=True)
    
    # Чтение растрового изображения средствами OpenCV
    img = cv2.imread(str(img_path))
    if img is None:
        print("СБОЙ [Ошибка декодирования файла]", flush=True)
        continue

    try:
        # Запуск нейросетевого подавления артефактов и апскейла.
        # outscale=2 определяет финальный коэффициент масштабирования выходной матрицы пикселей
        output, _ = upscaler.enhance(img, outscale=2)
        
        # Формирование пути экспорта и запись восстановленного массива данных на диск
        out_path = OUTPUT_DIR / f"ai_{img_path.name}"
        cv2.imwrite(str(out_path), output)
        print("УСПЕШНО", flush=True)
        
    except Exception as e:
        print(f"ОТКЛОНЕНО [Исключение среды выполнения: {e}]", flush=True)

print(f"\n[ЗАВЕРШЕНО]: Пакетный инференс успешно выполнен. Выходной каталог: {OUTPUT_DIR}", flush=True)
```

## 2. Анализ коллизии пространств имен (Namespace Collision)
При стандартной установке зависимостей из манифеста modules/NAFNet/requirements.txt пакетный менеджер pip загружает официальный релиз фреймворка BasicSR из репозитория PyPI, размещая его в глобальном каталоге среды (.venv/lib/python3.12/site-packages/basicsr).
## Суть проблемы (Root Cause)

   1. Различие реализаций: Авторы NAFNet глубоко модифицировали исходный код BasicSR под свои задачи и добавили фабричный метод create_model в локальный подкаталог modules/NAFNet/basicsr. В официальной open-source версии фреймворка этой функции не существует (там используется метод build_model).
   2. Конфликт импорта: При вызове инструкции from basicsr.models import create_model интерпретатор Python в первую очередь обращается к глобальному кэшу пакетов sys.modules и директории site-packages. Обнаружив там стандартный basicsr, Python блокирует импорт и генерирует критическое исключение:
   ImportError: cannot import name 'create_model' from 'basicsr.models'.
   3. Неэффективность манипуляций с sys.path: Использование sys.path.insert(0, ...) в рантайме не решает проблему, если верхнеуровневый пакет basicsr уже был единожды инициализирован и закэширован подсистемой импорта.

## Решение: Сборка пакета в режиме разработчика (Editable Mode)
Единственным легитимным решением является полное удаление внешнего пакета и принудительная регистрация локальной папки субмодуля в качестве единственного источника пространства имен basicsr.
